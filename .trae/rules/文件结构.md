# 项目文件结构

```
precastconcrete-qc/
├── .trae/rules/              # 项目规则与文档
├── .vscode/                  # VSCode 配置
│   ├── launch.json
│   ├── settings.json
│   └── tasks.json
├── config/                   # 权重/参数配置
│   └── feature_weights.json  # 点云聚类特征权重
├── fit/                      # 拟合算法模块
│   ├── fitters.py            # concrete_fit / rebar_fit
│   ├── get_part.py           # 提取混凝土/钢筋子点云
│   ├── io.py                 # 保存平面/直线参数
│   ├── k_means.py            # K-means 聚类
│   └── neighbor_features.py  # 邻域特征计算
├── geometry/                 # 几何数据类型与计算
│   ├── __init__.py           # 导出所有几何类
│   ├── base_utils.py         # 基础几何工具
│   ├── estimate.py           # 平面/直线估计与内点判定
│   ├── geo_utils.py          # 几何工具函数
│   ├── line.py               # BaseLine / CrossLine / SegmentLine
│   ├── plane.py              # BasePlane
│   ├── point.py              # BasePoint
│   └── utils.py              # 投影、相交等计算
├── models/                   # 拟合器（RANSAC）
│   ├── __init__.py
│   ├── adj_k_means.py        # 自适应 K-means
│   ├── line_fitter.py        # RansacLineFitter
│   └── plane_fitter.py       # RansacPlaneFitter
├── rebuild/                  # 三维拓扑重建
│   ├── concrete_params.py    # 混凝土交线/交点计算
│   ├── rebar_params.py       # 钢筋参数提取
│   └── topo_reconstr.py      # 面构建、网格生成
├── utils/                    # 工具模块
│   ├── eigen_measures_utils.py
│   ├── format_utils.py
│   ├── noise_utils.py        # 高斯噪声模拟
│   ├── pcd_utils.py          # OBB 内接圆柱计算
│   ├── test_utils.py
│   └── visualization_utils.py
├── mock/                     # 测试数据与脚本
│   ├── res/                  # 点云文件 (.ply)
│   ├── Revit2Point.md        # Revit→点云教程
│   └── test.py
├── main.py                   # 主流程入口
├── debug.py                  # 调试脚本
├── test.py                   # 测试脚本
├── pyproject.toml
└── README.md
```
