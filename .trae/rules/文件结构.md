# 项目文件结构

```
precastconcrete-qc/
├── .trae/rules/              # 项目规则与文档
├── .vscode/                  # VSCode 配置
├── docs/                     # 参考资料 (PDF)
├── experiments/              # 实验脚本 (15个实验 + 1共享模块)
│   ├── _common.py            # 共享训练模块 (种子/常量/训练循环/验证循环)
│   ├── experiment_pointnet2_baseline.py  # PointNet++ baseline
│   ├── experiment_pointnet_baseline.py   # PointNet 基线
│   ├── experiment_pointmlp_baseline.py   # PointMLP 基线
│   ├── experiment_pointnext_baseline.py  # PointNeXt 基线
│   ├── experiment_sota_comparison.py     # SOTA 综合对比
│   ├── experiment_benchmark_comparison.py# 基准对比
│   ├── experiment_a1_no_geo.py           # [消融] 去掉几何特征
│   ├── experiment_a2_fixed_ce.py         # [消融] 固定交叉熵权重
│   ├── experiment_a3a_noise_ignore.py    # [消融] 忽略噪声类
│   ├── experiment_a3b_noise_equal.py     # [消融] 噪声类等权
│   ├── experiment_a4_no_filter.py        # [消融] 无过滤
│   ├── experiment_s1_patch_radius.py     # [鲁棒] 邻域半径
│   ├── experiment_s3_density.py          # [鲁棒] 点密度
│   └── experiment_s4_noise.py            # [鲁棒] 噪声水平
├── lib/                      # 算法库
│   └── pointnet2/            # 点云分割模块
│       ├── __init__.py
│       ├── model.py          # PointNet++ 模型 (SA/FP 层)
│       ├── dataset.py        # 数据集加载与预处理
│       ├── metrics.py        # IoU/Precision/Recall/F1 评估指标
│       ├── pointnet_model.py     # PointNet 原始模型
│       ├── pointmlp_model.py     # PointMLP 模型
│       └── pointnext_model.py    # PointNeXt 模型
├── mock/                     # 合成点云数据
│   ├── generate_synthetic_dataset.py  # 数据生成器 (T梁/矩形梁/柱/L形柱/叠合梁)
│   ├── preview_mock.py                # Open3D 交互式预览
│   └── res/point/            # NPZ 数据文件
│       ├── augmented/        # 增强数据 (噪声×密度组合, 训练/验证用)
│       └── original/         # 原始干净数据 + JSON 元数据 (验证/测试)
├── checkpoints/              # 模型训练输出 (自动创建)
├── scripts/                  # 辅助脚本
├── .venv/                    # Python 虚拟环境
├── .python-version
├── pyproject.toml
└── .gitignore
```
