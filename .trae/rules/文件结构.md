# 项目文件结构

```
precastconcrete-qc/
├── .trae/rules/              # 项目规则与文档
├── .vscode/                  # VSCode 配置 (launch/settings/tasks)
├── docs/                     # 论文文档 (章节MD + PDF参考资料)
│   ├── pdfs/                 # 参考文献 PDF
│   ├── 1 引言.md
│   ├── 2 方法论.md
│   ├── 3 合成数据集构建.md
│   ├── 4 基线对比实验.md
│   ├── 5 消融实验与敏感度分析.md
│   ├── 6 结论.md
│   ├── pipeline-overview.md
│   └── refs.bib
├── experiments/              # 实验层 (独立训练脚本 + 训练器)
│   ├── _config.py            # 架构注册表、默认超参数
│   ├── _s1_trainer.py        # S1 训练器 (3类分割, 支持多架构)
│   ├── _s2_trainer.py        # S2 训练器 (级联精分割, S1冻结→预计算→训练)
│   ├── _a2_trainer.py        # A2 消融训练器 (单阶段4类)
│   ├── experiment_phase1_baseline.py # Phase 1: S1 三分类入口
│   ├── experiment_phase2_baseline.py # Phase 2: S2 精分割入口
│   ├── experiment_a1a_random_sampling.py  # [消融] 纯随机采样
│   ├── experiment_a1b_no_pos_balance.py   # [消融] 无正样本平衡
│   ├── experiment_a2_cascade_ablation.py  # [消融] 单阶段4类
│   ├── experiment_s1_patch_radius.py  # [敏感] 球查询半径
│   ├── experiment_s2_voxel_size.py    # [敏感] S2体素大小
│   ├── experiment_s3_density.py       # [敏感] 点云密度
│   ├── experiment_s4_noise.py         # [敏感] 噪声水平
│   └── checkpoints/          # 模型权重输出 (自动创建)
│       ├── phase1_baseline_{arch}/
│       │   ├── best_s1.pth
│       │   ├── history_phase1_baseline.json
│       │   └── summary.json
│       ├── phase2_baseline_{arch}/
│       │   ├── best_s2.pth
│       │   ├── s2_train.pt / s2_val.pt / s2_test.pt  (S2预计算缓存)
│       │   └── summary.json
│       ├── ablation_A2/
│       └── sensitivity_*/
├── lib/                      # 算法库 (三层)
│   ├── backbones/            # 骨架模型 (4种架构)
│   │   ├── __init__.py       # 公开API
│   │   ├── _base.py          # 基础构建块 (SA/FP/FPS/BallQuery/FocalLoss/WeightedCE)
│   │   ├── pointnet2.py      # PointNet++ + GatedPointNet++
│   │   ├── pointnet.py       # PointNet (无T-Net)
│   │   ├── pointmlp.py       # PointMLP (GeometricAffine + ResMLP)
│   │   └── pointnext.py      # PointNeXt
│   ├── cascade/              # 级联管线方法
│   │   ├── __init__.py
│   │   ├── filter.py         # 自适应体素降采样 (仅混凝土降采样)
│   │   ├── stage2_model.py   # S2模型 (PointNet++ 6ch输入)
│   │   ├── precompute.py     # S2预计算与缓存 (S1推理→过滤→磁盘)
│   │   └── validate.py       # 级联管线验证 (V2全管线+S2缓存版)
│   ├── utils/                # 共享工具集
│   │   ├── __init__.py
│   │   ├── common.py         # 全局常量/设备/种子/标签/路径
│   │   ├── datasets.py       # 数据加载器 (GpuCachedTrainLoader/Val/Test)
│   │   ├── metrics.py        # IoU/Precision/Recall/F1
│   │   ├── training.py       # 泛型训练/验证循环
│   │   └── history.py        # 历史记录/Checkpoint管理
│   └── pointcloud_generator/ # 点云生成器 (合成数据)
│       ├── __init__.py       # 公开API
│       ├── __main__.py       # 命令行入口
│       ├── _builder.py       # 构件builder (7种构件)
│       ├── _concrete.py      # 混凝土棱柱采样、挖孔
│       ├── _config.py        # 标签定义、参数常量、种子工具
│       ├── _dataset.py       # 批量数据集生成器
│       ├── _geometry.py      # 几何工具 (多边形/圆柱/法线)
│       ├── _noise.py         # 噪声增强 (高斯/离群/缺失/边缘)
│       ├── _rebar.py         # 钢筋生成 (纵筋/箍筋)
│       └── _sleeve.py        # 灌浆套筒生成
├── mock/                     # 合成点云数据 (NPZ)
│   ├── origin/               # 原始生成 (clean)
│   │   └── Mock_{type}_{id}_{variant}_{noise}_{density}.npz
│   ├── add_noise/            # 噪声增强版本
│   └── inuse/                # 当前在用数据子集 (6:2:2划分)
│       ├── train/
│       ├── val/
│       └── test/
├── scripts/
│   └── query_openalex.py     # OpenAlex API查询辅助脚本
├── .venv/                    # Python 虚拟环境
├── .python-version
├── pyproject.toml            # 项目元数据 + 依赖声明
├── uv.lock
├── uv.toml
└── .gitignore
```
