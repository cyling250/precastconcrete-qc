# 项目文件结构

```
precastconcrete-qc/
├── .trae/rules/              # 项目规则与文档
├── .vscode/                  # VSCode 配置
├── docs/                     # 参考资料 (PDF / 方法论文档)
├── experiments/              # 实验脚本 (10个)
│   ├── _common.py                    # 共享工具 (常量/种子/数据加载器/训练/验证)
│   ├── experiment_phase1_baseline.py # Phase 1: S1 三分类 (噪声/混凝土/金属), + --arch CLI
│   ├── experiment_phase2_baseline.py # Phase 2: S2 四类精分割, + --arch CLI
│   ├── experiment_a1a_random_sampling.py  # [消融] 纯随机采样替代局部块采样
│   ├── experiment_a1b_no_pos_balance.py   # [消融] 无正样本平衡 (pos_ratio=1.0)
│   ├── experiment_a2_cascade_ablation.py  # [消融] 单阶段 4 类 (无级联)
│   ├── experiment_s1_patch_radius.py  # [敏感] 球查询半径 (100/150/200/300mm)
│   ├── experiment_s2_voxel_size.py    # [敏感] S2 体素大小 (20/40/80mm)
│   ├── experiment_s3_density.py       # [敏感] 点云密度评估
│   └── experiment_s4_noise.py         # [敏感] 噪声水平评估
├── lib/                      # 算法库
│   ├── pointcloud_generator/ # 点云生成器 (构件级合成数据)
│   │   ├── __init__.py       # 公开 API (make_point_cloud, builders, DatasetGenerator)
│   │   ├── __main__.py       # 命令行入口
│   │   ├── _builder.py       # 构件 builder (7 种构件类型)
│   │   ├── _concrete.py      # 混凝土棱柱采样、挖孔
│   │   ├── _config.py        # 标签定义、常量、种子工具
│   │   ├── _dataset.py       # 批量数据集生成器 DatasetGenerator
│   │   ├── _geometry.py      # 几何工具 (多边形操作、圆柱采样)
│   │   ├── _noise.py         # 噪声增强 (高斯/离群点/缺失块)
│   │   ├── _rebar.py         # 钢筋生成 (纵筋、箍筋)
│   │   └── _sleeve.py        # 灌浆套筒生成
│   └── pointnet2/            # 点云分割模块
│       ├── __init__.py
│       ├── model.py          # PointNet++ 模型 (SA/FP 层, PerSampleWeightedCE, FocalLoss)
│       ├── dataset.py        # 数据集加载 (GpuCachedTrainLoader, RandomValDataset, RandomSampleDataset)
│       ├── metrics.py        # IoU/Precision/Recall/F1 评估指标
│       ├── pointnet_model.py     # PointNet 模型
│       ├── pointmlp_model.py     # PointMLP 模型
│       └── pointnext_model.py    # PointNeXt 模型
├── mock/                     # 合成点云数据 (NPZ)
│   ├── add_noise/            # 噪声增强版本 (训练/验证用)
│   │   └── Mock_{type}_{id}_{variant}_{noise}_{density}.npz
│   └── inuse/                # 当前在用数据子集
│       ├── train/            # 训练集
│       ├── val/              # 验证集
│       └── test/             # 测试集
├── checkpoints/              # 模型训练输出 (自动创建)
├── scripts/                  # 辅助脚本
├── .venv/                    # Python 虚拟环境
├── .python-version
├── pyproject.toml
├── uv.lock
├── uv.toml
└── .gitignore
```
